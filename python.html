8) Files (read/write)

9) Errors & exceptions

Use try/except to handle failures, and finally for cleanup.

Level: mid
Example: handle invalid input
def to_int(text):
    try:
        return int(text)
    except ValueError:
        return None

print(to_int("123"))
print(to_int("hello"))
Expected output
123
None
Example: multiple except + else + finally
def safe_div(a, b):
    try:
        result = a / b
    except ZeroDivisionError:
        return "Cannot divide by zero"
    except TypeError:
        return "Inputs must be numbers"
    else:
        return result
    finally:
        # Runs no matter what (useful for cleanup)
        pass

print(safe_div(10, 2))
print(safe_div(10, 0))
print(safe_div("10", 2))
Expected output
5.0
Cannot divide by zero
Inputs must be numbers

Good practice: catch specific exceptions (like ValueError) instead of a broad except:.

11) OOP basics (classes)

Make a small class with state, methods, and a readable string representation.

Level: mid

When to use a class: you have data + behavior that belong together (like a “BankAccount” or “TodoList”).

Example: class with methods
class Counter:
    def __init__(self, start=0):
        self.value = start

    def inc(self, step=1):
        self.value += step
        return self.value

    def reset(self):
        self.value = 0

    def __repr__(self):
        return f"Counter(value={self.value})"

c = Counter()
print(c)
c.inc()
c.inc(4)
print(c.value)
c.reset()
print(c)
Expected output
Counter(value=0)
5
Counter(value=0)
Example: @property (clean attribute access)
class Rectangle:
    def __init__(self, w, h):
        self.w = w
        self.h = h

    @property
    def area(self):
        return self.w * self.h

r = Rectangle(3, 4)

12
Practice (write code)
  1. Create a Student class with name and grades (list). Add a method average().
  2. Add __repr__ so printing the object is informative.
  3. Create two students and compare their averages.
  4. Add a property is_passing that returns True if average ≥ 60.

class Student:
def __init__(self, name, grades):
self.name = name
self.grades = grades

def average(self):
return sum(self.grades) / len(self.grades) if self.grades else 0

def __repr__(self):
return f"Student({self.name!r}, avg={self.average():.2f})"

@property
def is_passing(self):
return self.average() >= 60

def add_grade(self, grade):
self.grades.append(grade)

# Usage
s1 = Student("Alice", [70, 80, 90])
s2 = Student("Bob", [50, 55, 60])
print(s1)
print(s2)
print(s1.is_passing, s2.is_passing)
s2.add_grade(70)
print(s2.average())
